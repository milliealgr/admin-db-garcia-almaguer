Table clientes {
  dni_cliente varchar [primary key, not null]
  nombre_completo varchar
  telefono varchar
  correo varchar
}

Table mascotas {
  id_mascota int [primary key, increment, not null]
  nombre varchar
  especie varchar
  raza varchar
  fecha_nacimiento date
  dni_cliente varchar [not null]
}

Table veterinarios {
  dni_veterinario varchar [primary key, not null]
  nombre_completo varchar
  especialidad varchar
  telefono varchar
}

Table atenciones_medicas {
  id_atencion int [primary key, increment, not null]
  fecha_hora datetime
  diagnostico varchar
  costo_base decimal
  id_mascota int [not null]
  dni_veterinario varchar [not null]
}

Table medicamentos {
  codigo_medicamento int [primary key, not null]
  nombre_comercial varchar
  laboratorio varchar
  precio_unitario decimal
}

Table prescripciones {
  id_prescripcion int [primary key, increment, not null]
  id_atencion int [not null]
  codigo_medicamento int [not null]
  cantidad int
  indicaciones varchar
}

Ref: mascotas.dni_cliente > clientes.dni_cliente
Ref: atenciones_medicas.id_mascota > mascotas.id_mascota
Ref: atenciones_medicas.dni_veterinario > veterinarios.dni_veterinario
Ref: prescripciones.id_atencion > atenciones_medicas.id_atencion
Ref: prescripciones.codigo_medicamento > medicamentos.codigo_medicamento
